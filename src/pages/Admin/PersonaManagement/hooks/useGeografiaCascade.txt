import { useState, useCallback, useRef, useEffect } from "react";
import geografiaService from "../../../../services/geografiaService";
import nacionService from "../../../../services/nacionService";
import { esNacionArgentina } from "../utils/nacionUtils";

/**
 * Normaliza la respuesta de los servicios geográficos:
 * soporta `r.data.data`, `r.data` o `r` directamente como array.
 */
const normalizeList = (r) => r?.data?.data || r?.data || r || [];

/**
 * Hook que maneja la cascada geográfica: Provincias -> Departamentos -> Localidades.
 * Incluye tokens independientes por entidad para evitar race conditions
 * y una caché en memoria para resolver búsquedas concurrentes e instantáneas.
 */
const useGeografiaCascade = () => {
  const [nacions, setNacions] = useState([]);
  const [provincias, setProvincias] = useState([]);
  const [departamentos, setDepartamentos] = useState([]);
  const [localidades, setLocalidades] = useState([]);

  // Caché en memoria durante el ciclo de vida de la pantalla
  const cacheDeptos = useRef(new Map());
  const cacheLocs = useRef(new Map());

  // Tokens de petición independientes por nivel de cascada
  const deptoReqId = useRef(0);
  const locReqId = useRef(0);

  // Carga inicial de provincias (flujo de nacimiento de PersonaFormModal).
  useEffect(() => {
    let active = true;
    geografiaService
      .getProvincias()
      .then((r) => {
        if (active) setProvincias(normalizeList(r));
      })
      .catch((error) => {
        console.error("Error al cargar provincias:", error);
      });
    return () => {
      active = false;
    };
  }, []);

  // Carga el catálogo de naciones/países.
  useEffect(() => {
    let active = true;
    nacionService
      .getAll({ per_page: 1000, search: "" })
      .then((r) => {
        if (active) setNacions(normalizeList(r));
      })
      .catch((error) => {
        console.error("Error al cargar naciones:", error);
      });
    return () => {
      active = false;
    };
  }, []);

  const loadDepartamentos = useCallback(async (provinciaId) => {
    if (!provinciaId) {
      setDepartamentos([]);
      return;
    }

    // Servir desde la caché si ya existe
    if (cacheDeptos.current.has(String(provinciaId))) {
      setDepartamentos(cacheDeptos.current.get(String(provinciaId)));
      return;
    }

    const currentId = ++deptoReqId.current;
    try {
      const r = await geografiaService.getDepartamentos(provinciaId);
      const data = normalizeList(r);
      cacheDeptos.current.set(String(provinciaId), data);

      if (currentId === deptoReqId.current) {
        setDepartamentos(data);
      }
    } catch (error) {
      console.error("Error al cargar departamentos:", error);
    }
  }, []);

  const loadLocalidades = useCallback(async (departamentoId) => {
    if (!departamentoId) {
      setLocalidades([]);
      return;
    }

    // Servir desde la caché si ya existe
    if (cacheLocs.current.has(String(departamentoId))) {
      setLocalidades(cacheLocs.current.get(String(departamentoId)));
      return;
    }

    const currentId = ++locReqId.current;
    try {
      const r = await geografiaService.getLocalidades(departamentoId);
      const data = normalizeList(r);
      cacheLocs.current.set(String(departamentoId), data);

      if (currentId === locReqId.current) {
        setLocalidades(data);
      }
    } catch (error) {
      console.error("Error al cargar localidades:", error);
    }
  }, []);

  const clearCascade = useCallback(() => {
    ++deptoReqId.current;
    ++locReqId.current;
    setDepartamentos([]);
    setLocalidades([]);
  }, []);

  // Vacía la cascada geográfica argentina (provincias/departamentos/localidades).
  const clearGeoArgentina = useCallback(() => {
    ++deptoReqId.current;
    ++locReqId.current;
    setProvincias([]);
    setDepartamentos([]);
    setLocalidades([]);
  }, []);

  const handleProvinciaChange = useCallback(
    async (provinciaId) => {
      setDepartamentos([]);
      setLocalidades([]);
      if (!provinciaId) return;
      await loadDepartamentos(provinciaId);
    },
    [loadDepartamentos],
  );

  const handleDepartamentoChange = useCallback(
    async (departamentoId) => {
      setLocalidades([]);
      if (!departamentoId) return;
      await loadLocalidades(departamentoId);
    },
    [loadLocalidades],
  );

  const handleNacionChange = useCallback(
    (nacionId) => {
      if (!nacionId || !esNacionArgentina(nacions, nacionId)) {
        clearGeoArgentina();
        return;
      }
      // Argentina: volver a cargar provincias (idempotente).
      geografiaService
        .getProvincias()
        .then((r) => setProvincias(normalizeList(r)))
        .catch((error) => console.error("Error al cargar provincias:", error));
    },
    [nacions, clearGeoArgentina],
  );

  return {
    nacions,
    provincias,
    departamentos,
    localidades,
    isNacionArgentinaValue: (nacionId) => esNacionArgentina(nacions, nacionId),
    loadDepartamentos,
    loadLocalidades,
    clearCascade,
    clearGeoArgentina,
    handleProvinciaChange,
    handleDepartamentoChange,
    handleNacionChange,
  };
};

export default useGeografiaCascade;
